# v0.4 Phase 1 — Filtered Transcript + Tool-Call Summary Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. **Each substantive task ends with a Ralph reviewer stack** (GPT-5 via `llm -t review-code`, `debugging-toolkit:debugger`, `code-refactoring:code-reviewer`, `llm-application-dev:ai-engineer`) before the next starts.

**Goal:** Replace the raw 5 KB transcript tail polymath currently passes to lens analysts with (a) a filtered conversation containing only user messages + Claude's visible text + thinking blocks, and (b) a structured one-line-per-call summary of recent tool invocations. Net effect: ~3× input tokens, ~5× signal density, hallucination rate flat or down.

**Architecture:** Two new pure-bash library modules — `lib/transcript.sh` (JSONL filter using jq) and `lib/tool-summary.sh` (renderer for the structured tool-call block) — sourced from `bin/statusline.sh`. The analyst-primary prompt gains two new placeholders (`${transcript_filtered}` and `${tool_summary}`), registered in `lib/prompt-loader.sh`'s allowlist so the single-pass substitution invariant is preserved. Existing `lib/grounding.sh` redaction patterns run over both new fields before injection — no new secret-leak surface.

**Tech Stack:** bash 3.2-portable (no mapfile, no `${var,,}`, no `local -n`, no flock); jq for JSONL parsing (already a project dep); bats for tests (already gates CI); `LC_ALL=C` discipline on numeric sort/awk (file-top convention).

**Branch:** `feat/v0.4-filtered-transcript` from `origin/main` @ `2dc41c5`.

**Eval gate to merge:** `bats test/` green, `shellcheck` clean, eval-gate shows useful% ≥+5pp on ≥4 of 7 lenses with hallucination% non-regressing, p95 cycle time ≤ pre-change baseline +500ms.

#!/usr/bin/env bash
# lib/transcript.sh — filtered conversation extractor for analyst context.
#
# Owns: the transformation from a raw Claude Code JSONL transcript into a
# compact "what humans said + what Claude said + what Claude thought"
# stream. Drops tool inputs and outputs because (a) they're huge, and
# (b) the structured tool-call summary in lib/tool-summary.sh covers the
# same signal with bounded cost.
#
# Public functions:
#   pp_transcript_filter <path>       → stdout: filtered text
#   pp_transcript_tool_calls <path>   → stdout: JSON array of tool calls
#
# Env vars (defaults in config/default.env):
#   PP_TRANSCRIPT_MAX               (16384) Output cap, tail-biased
#   PP_TRANSCRIPT_USER_TRUNC        (2048)  Per-user-message cap
#   PP_TRANSCRIPT_ASSISTANT_TRUNC   (4096)  Per-assistant-text cap
#   PP_TRANSCRIPT_KEEP_THINKING     (1)     1=keep, 0=drop
#   PP_TRANSCRIPT_TOOL_TRUNC        (512)   Drop tool blocks > this size
#
# Bash 3.2-portable. Uses jq for parsing. Silently no-ops on missing file.
# Malformed JSON lines are skipped (jq -e per-line gate).

# Source-guard: re-source is a no-op.
if [ -n "${_PP_TRANSCRIPT_SOURCED:-}" ]; then return 0; fi
_PP_TRANSCRIPT_SOURCED=1

# Numerically stable across locales (decimal separator).
LC_ALL=C

: "${PP_TRANSCRIPT_MAX:=16384}"
: "${PP_TRANSCRIPT_USER_TRUNC:=2048}"
: "${PP_TRANSCRIPT_ASSISTANT_TRUNC:=4096}"
: "${PP_TRANSCRIPT_KEEP_THINKING:=1}"
: "${PP_TRANSCRIPT_TOOL_TRUNC:=512}"

# pp_transcript_filter <path>
# Stdout: filtered conversation text, one event per line, tail-biased.
# Format:
#   USER: <message>
#   CLAUDE: <text>
#   CLAUDE(thinking): <text>          (only when PP_TRANSCRIPT_KEEP_THINKING=1)
# Empty output for non-existent / empty / unreadable input.
pp_transcript_filter() {
  local _path="${1:-}"
  if [ -z "$_path" ] || [ ! -r "$_path" ]; then
    return 0
  fi
  if ! command -v jq >/dev/null 2>&1; then
    return 0
  fi

  # jq program:
  #   - Per-line: select user OR assistant.
  #   - User: emit "USER: <content>" if content is a STRING (raw user msg).
  #           Skip if content is an array (those are tool_result wrappers).
  #   - Assistant: walk content[] and emit one line per text or thinking block.
  #   - Skip tool_use and tool_result entirely (the summary module covers them).
  # The -R option on jq lets us read line-by-line with error tolerance;
  # we re-parse each line inside the program. Bad lines yield empty output.
  local _keep_think="${PP_TRANSCRIPT_KEEP_THINKING}"
  local _user_cap="${PP_TRANSCRIPT_USER_TRUNC}"
  local _ast_cap="${PP_TRANSCRIPT_ASSISTANT_TRUNC}"

  local _jq_prog
  _jq_prog='
    def trunc($n): if (. | length) > $n then (.[0:$n] + "...[truncated]") else . end;
    . as $raw
    | (try ($raw | fromjson) catch null) as $rec
    | select($rec != null)
    | if $rec.type == "user" then
        ($rec.message.content // null) as $c
        | if ($c | type) == "string" then
            "USER: " + ($c | trunc('"$_user_cap"'))
          else empty end
      elif $rec.type == "assistant" then
        ($rec.message.content // []) as $arr
        | $arr[]
        | if .type == "text" then
            "CLAUDE: " + ((.text // "") | trunc('"$_ast_cap"'))
          elif .type == "thinking" and '"$_keep_think"' == 1 then
            "CLAUDE(thinking): " + ((.text // "") | trunc('"$_ast_cap"'))
          else empty end
      else empty end
  '

  # Full filtered text first, then tail-clip to PP_TRANSCRIPT_MAX.
  # `tail -c` is byte-accurate on macOS + GNU; both implementations
  # already used elsewhere in the codebase.
  local _full
  _full=$(jq -Rr "$_jq_prog" "$_path" 2>/dev/null)
  if [ -z "$_full" ]; then
    return 0
  fi
  printf '%s\n' "$_full" | tail -c "$PP_TRANSCRIPT_MAX"
}

# pp_transcript_tool_calls <path>
# Stdout: JSON array of tool invocations (newest at end).
# Each element: {tool, target, exit, summary} where:
#   tool   = "Read"|"Edit"|"Write"|"Bash"|"Grep"|"WebFetch"|...
#   target = file path / command / pattern
#   exit   = exit code if tool_result present, else null
#   summary = first 120 chars of tool_result text, null if missing
# Returns "[]" on empty/missing input. Bounded by PP_TOOL_SUMMARY_MAX (set elsewhere).
pp_transcript_tool_calls() {
  local _path="${1:-}"
  if [ -z "$_path" ] || [ ! -r "$_path" ]; then
    printf '[]'
    return 0
  fi
  if ! command -v jq >/dev/null 2>&1; then
    printf '[]'
    return 0
  fi

  # Slurp the JSONL (jq -s) so we can correlate tool_use → tool_result
  # by sibling position. tool_result entries come back as type=user with
  # content[].type=tool_result in Claude Code's transcript schema.
  jq -s '
    def shorten($n): if (. | length) > $n then (.[0:$n] + "...") else . end;
    # Build a flat list of tool events keyed by position.
    [ .[]
      | select(.type == "assistant" or .type == "user")
      | .message.content as $c
      | if ($c | type) == "array" then
          $c[]
          | select(.type == "tool_use" or .type == "tool_result")
        else empty end
    ]
    | . as $events
    | [ range(0; length)
        | . as $i
        | $events[$i] as $e
        | select($e.type == "tool_use")
        | {
            tool:   ($e.name // "?"),
            target: ($e.input.file_path // $e.input.command // $e.input.pattern // $e.input.url // ""),
            exit:   null,
            summary: null,
          } as $base
        | if ($i + 1) < ($events | length) and $events[$i+1].type == "tool_result"
          then
            $events[$i+1] as $r
            | $base + {
                summary: (
                  ($r.content // [] | map(select(.type == "text") | .text) | join(" ")) | shorten(120)
                )
              }
          else $base end
      ]
  ' "$_path" 2>/dev/null || printf '[]'
}

```bash
#!/usr/bin/env bash
# lib/tool-summary.sh — render a tool-call JSON array as a human-readable
# one-line-per-call block for analyst context.
#
# Owns: the JSON → text rendering for the lens's "what did Claude actually
# DO since last cycle" signal. Bounded by PP_TOOL_SUMMARY_MAX and runs
# every string through pp_redact_secrets if available (so a malicious
# file path can't smuggle credentials into the LLM call).
#
# Public function:
#   pp_tool_summary_render <json_array_string>   → stdout: formatted block
#
# Env:
#   PP_TOOL_SUMMARY_MAX   (20)   Max lines kept (tail-biased: newest stay)
#
# Bash 3.2-portable. jq + project's existing redaction.

if [ -n "${_PP_TOOL_SUMMARY_SOURCED:-}" ]; then return 0; fi
_PP_TOOL_SUMMARY_SOURCED=1

LC_ALL=C
: "${PP_TOOL_SUMMARY_MAX:=20}"

# Try to source grounding.sh for pp_redact_secrets. If unavailable
# (e.g. when called from a test that didn't source it), fall back to
# a built-in mini-redactor that handles the canonical key shapes.
if [ -z "${_PP_GROUNDING_SOURCED:-}" ]; then
  _pp_ts_self_dir="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
  if [ -r "${_pp_ts_self_dir}/grounding.sh" ]; then
    # shellcheck source=grounding.sh
    . "${_pp_ts_self_dir}/grounding.sh" 2>/dev/null || true
  fi
fi

# Fallback redactor — used only if lib/grounding.sh wasn't loaded.
_pp_ts_redact_fallback() {
  printf '%s' "$1" \
    | sed -E 's/sk-[A-Za-z0-9_-]{20,}/[REDACTED-OPENAI]/g' \
    | sed -E 's/AKIA[0-9A-Z]{16}/[REDACTED-AWS]/g' \
    | sed -E 's/ghp_[A-Za-z0-9]{30,}/[REDACTED-GH]/g' \
    | sed -E 's/eyJ[A-Za-z0-9_-]{20,}\.[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}/[REDACTED-JWT]/g'
}

_pp_ts_redact() {
  if command -v pp_redact_secrets >/dev/null 2>&1; then
    printf '%s' "$1" | pp_redact_secrets
  else
    _pp_ts_redact_fallback "$1"
  fi
}

# pp_tool_summary_render <json_string>
pp_tool_summary_render() {
  local _json="${1:-}"
  if [ -z "$_json" ]; then
    printf '(no recent tool calls)'
    return 0
  fi
  if ! command -v jq >/dev/null 2>&1; then
    printf '(jq unavailable)'
    return 0
  fi

  # Validate the JSON; on parse fail, return the no-op string.
  if ! printf '%s' "$_json" | jq -e 'type == "array"' >/dev/null 2>&1; then
    printf '(no recent tool calls)'
    return 0
  fi
  local _len
  _len=$(printf '%s' "$_json" | jq 'length' 2>/dev/null || echo 0)
  if [ "$_len" = "0" ]; then
    printf '(no recent tool calls)'
    return 0
  fi

  # Tail-bias: keep last PP_TOOL_SUMMARY_MAX entries.
  local _trimmed
  _trimmed=$(printf '%s' "$_json" | jq --argjson n "$PP_TOOL_SUMMARY_MAX" '
    if length > $n then .[length - $n:] else . end
  ')

  # Format. printf "%-6s %s" gives a 6-char tool-name column.
  local _lines
  _lines=$(printf '%s' "$_trimmed" | jq -r '
    .[]
    | (
        ((.tool // "?") | .[0:6])
        + "   "
        + (.target // "")
        + (if .summary != null and .summary != "" then "    | " + .summary else "" end)
      )
  ')

  # Run the whole rendered block through the redactor — protects against
  # secret-shaped paths and tool-result snippets.
  _pp_ts_redact "$_lines"
}
```
