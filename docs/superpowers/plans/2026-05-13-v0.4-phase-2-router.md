# v0.4 Phase 2 — Router Meta-Lens Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. **Each substantive task ends with a Ralph reviewer stack** (GPT-5 via `llm -t review-code`, `debugging-toolkit:debugger`, `code-refactoring:code-reviewer`, `llm-application-dev:ai-engineer`) before the next starts.

**Goal:** Replace the fan-out-to-all-7-lenses fixed pattern with a router meta-lens that picks **1–3 lenses per cycle** based on conversation signals (phase / confidence / outcome / tone). Net effect: ~60% reduction in analyst LLM calls per cycle while increasing signal per observation (1 well-chosen voice beats 7 generic ones).

**Architecture:** Two-layer pick.

1. **Deterministic signal extractor** (`lib/router-signals.sh`) — no LLM. Parses filtered transcript + tool summary + cache_test_result + memory hints to produce a structured JSON: `{phase, confidence, outcome, tone, session_age_min, budget_remaining_pct, last_test_failed, recent_edit_density}`. Bash 3.2-portable, jq-based.
2. **LLM-based lens picker** (`lib/router.sh`) — one `gpt-5-mini` call (~$0.001). Reads the signals JSON + the enabled-lens registry (id + 1-line description), returns a NEWLINE-DELIMITED list of 1–`PP_ROUTER_MAX` lens IDs to fire.

**Plus a 20% surprise inject** (`PP_ROUTER_SURPRISE_PROB`): with the configured probability, append one random *off-discipline* lens to the picked set. Preserves the v0.3 serendipity property (the "security lens fires on a UI change") that uniform fan-out gave for free.

**Plus fail-open semantics:** if the router LLM call fails (timeout, bad JSON, budget exhausted, `PP_EVAL_MODE=1`), the fan-out falls back to ALL enabled lenses — same behavior as v0.3. Failure of the router never silences the system.

**Tech Stack:** bash 3.2-portable (no `mapfile`, no lowercase-parameter-expansion, no `local -n`, no `flock`, no global `LC_ALL=C`), jq for JSON I/O, `llm` CLI for the meta-call, bats for tests. NEWLINE-DELIMITED router output read via `while IFS= read -r` to avoid word-splitting on whitespace in lens IDs (per advisory).

**Branch:** `feat/v0.4-router` stacked on `feat/v0.4-filtered-transcript` (Phase 1). After Phase 1 PR merges, rebase onto `main`.

**Eval gate to merge:**
- `bats test/` green
- `shellcheck -S warning` clean on changed files
- Per-cycle cost reduction ≥30% measured against v0.3 baseline
- Routing accuracy ≥70% on 5 new fixtures where the correct lens is known
- No regression in non-router-mode (`PP_ROUTER_ENABLE=0` preserves v0.3 behavior byte-identically)

Create the file with deterministic regex/count-based detection:

```bash
#!/usr/bin/env bash
# lib/router-signals.sh — deterministic feature extraction for the router.

if [ -n "${_PP_ROUTER_SIGNALS_SOURCED:-}" ]; then return 0; fi
_PP_ROUTER_SIGNALS_SOURCED=1

# pp_router_extract_signals <transcript_filtered> <tool_summary_json>
# Stdout: JSON object with deterministic features.
pp_router_extract_signals() {
  local _tx="${1:-}"
  local _tools_json="${2:-[]}"

  # Normalize tool JSON; treat malformed as empty array.
  if ! printf '%s' "$_tools_json" | jq -e 'type == "array"' >/dev/null 2>&1; then
    _tools_json='[]'
  fi

  # ---- phase detection ----
  # Score the candidates; tie-break by priority order.
  local _phase="unknown"
  local _plan_hits _edit_count _read_count _bash_count _test_hits

  _plan_hits=$(printf '%s' "$_tx" | LC_ALL=C grep -ciE 'let me think|consider the|trade-?off|approach|should we|think about' 2>/dev/null || echo 0)
  _edit_count=$(printf '%s' "$_tools_json" | jq '[.[] | select(.tool == "Edit" or .tool == "Write" or .tool == "MultiE")] | length' 2>/dev/null)
  _read_count=$(printf '%s' "$_tools_json" | jq '[.[] | select(.tool == "Read")] | length' 2>/dev/null)
  _bash_count=$(printf '%s' "$_tools_json" | jq '[.[] | select(.tool == "Bash")] | length' 2>/dev/null)
  _test_hits=$(printf '%s' "$_tools_json" | jq '[.[] | select(.tool == "Bash" and (.target // "" | test("test|spec|ci"; "i")))] | length' 2>/dev/null)

  if [ "${_test_hits:-0}" -ge 1 ] && [ "${_bash_count:-0}" -ge 1 ]; then
    # Test-failure marker in summary?
    local _fail_marker
    _fail_marker=$(printf '%s' "$_tools_json" | jq '[.[] | select(.summary // "" | test("fail|FAIL|error|ERROR|✗|×"))] | length' 2>/dev/null)
    if [ "${_fail_marker:-0}" -ge 1 ]; then
      _phase="debugging"
    fi
  fi
  if [ "$_phase" = "unknown" ] && [ "${_edit_count:-0}" -gt "${_read_count:-0}" ]; then
    _phase="drafting"
  fi
  if [ "$_phase" = "unknown" ] && [ "${_plan_hits:-0}" -ge 1 ]; then
    _phase="planning"
  fi

  # ---- confidence ----
  local _hedges _definites _confidence="medium"
  _hedges=$(printf '%s' "$_tx" | LC_ALL=C grep -ciE 'i think|maybe|perhaps|might|not sure|could be|probably' 2>/dev/null || echo 0)
  _definites=$(printf '%s' "$_tx" | LC_ALL=C grep -ciE 'fixed|verified|works|completed|done|tested|confirmed' 2>/dev/null || echo 0)
  if [ "${_hedges:-0}" -gt "${_definites:-0}" ] && [ "${_hedges:-0}" -ge 2 ]; then
    _confidence="low"
  elif [ "${_definites:-0}" -gt "${_hedges:-0}" ] && [ "${_definites:-0}" -ge 2 ]; then
    _confidence="high"
  fi

  # ---- outcome ----
  local _outcome="unknown"
  local _fail_count
  _fail_count=$(printf '%s' "$_tools_json" | jq '[.[] | select(.summary // "" | test("FAIL|ERROR|exit 1|✗"; "i"))] | length' 2>/dev/null)
  local _pass_count
  _pass_count=$(printf '%s' "$_tools_json" | jq '[.[] | select(.summary // "" | test("passing|PASS|✓|ok"; "i"))] | length' 2>/dev/null)
  if [ "${_fail_count:-0}" -ge 1 ]; then
    _outcome="test_failed"
  elif [ "${_pass_count:-0}" -ge 1 ]; then
    _outcome="test_passed"
  fi

  # ---- tone ----
  local _frustration _tone="neutral"
  _frustration=$(printf '%s' "$_tx" | LC_ALL=C grep -ciE '^USER:.*\b(no|wrong|stop|why are you|that is wrong|not working|not getting)\b' 2>/dev/null || echo 0)
  if [ "${_frustration:-0}" -ge 2 ]; then
    _tone="frustrated"
  fi

  # ---- edit density (sliding window of recent tool calls) ----
  local _density="${_edit_count:-0}"

  # Compose JSON.
  jq -n \
    --arg phase "$_phase" \
    --arg confidence "$_confidence" \
    --arg outcome "$_outcome" \
    --arg tone "$_tone" \
    --argjson edit_density "${_density:-0}" \
    '{
      phase: $phase,
      confidence: $confidence,
      outcome: $outcome,
      tone: $tone,
      recent_edit_density: $edit_density
    }'
}
```

- [ ] **Step 4: Implement `lib/router.sh`**

Create the file. Key design points:
- `PP_ROUTER_FORCE_OUTPUT` is a test-only env that bypasses the LLM call (so tests don't need a real key).
- Output validation: lowercase, in enabled set, deduped.
- Floor (MIN) padding: append top-N enabled lenses if router under-returns.
- Cap (MAX): tail-clip if router over-returns.

```bash
#!/usr/bin/env bash
# lib/router.sh — LLM-based lens picker + surprise inject.

if [ -n "${_PP_ROUTER_SOURCED:-}" ]; then return 0; fi
_PP_ROUTER_SOURCED=1

: "${PP_ROUTER_ENABLE:=1}"
: "${PP_ROUTER_MAX:=3}"
: "${PP_ROUTER_MIN:=1}"
: "${PP_ROUTER_MODEL:=gpt-5-mini}"
: "${PP_ROUTER_TIMEOUT_S:=8}"
: "${PP_ROUTER_SURPRISE_PROB:=0.2}"
: "${PP_ROUTER_SURPRISE_MAX_TOTAL:=4}"

# pp_router_pick_lenses <signals_json> <transcript_filtered>
# Stdout: newline-delimited lens IDs (1..PP_ROUTER_MAX).
# Fail-open: on any error or when ENABLE=0 or EVAL_MODE=1, returns the
# full PP_LENS_IDS_AVAILABLE set (one per line).
pp_router_pick_lenses() {
  local _signals="${1:-}"
  local _tx="${2:-}"
  local _enabled="${PP_LENS_IDS_AVAILABLE:-}"

  # Emergency rollback / eval bypass.
  if [ "${PP_ROUTER_ENABLE:-1}" != "1" ] || [ "${PP_EVAL_MODE:-0}" = "1" ]; then
    printf '%s\n' $_enabled
    return 0
  fi

  # Test injection: skip LLM call when FORCE_OUTPUT is set.
  local _raw
  if [ -n "${PP_ROUTER_FORCE_OUTPUT+set}" ]; then
    _raw="$PP_ROUTER_FORCE_OUTPUT"
  else
    _raw=$(_pp_router_llm_call "$_signals" "$_tx" "$_enabled")
  fi

  # Filter to enabled set + dedupe (preserving order).
  local _seen="" _line _validated=""
  while IFS= read -r _line; do
    [ -z "$_line" ] && continue
    case " $_enabled " in
      *" $_line "*)
        case " $_seen " in
          *" $_line "*) ;;
          *)
            _validated="${_validated}${_line}"$'\n'
            _seen="$_seen $_line"
          ;;
        esac
      ;;
    esac
  done <<EOF
$_raw
EOF

  # Enforce floor: pad from enabled set if under MIN.
  local _count
  _count=$(printf '%s' "$_validated" | grep -c '^.' 2>/dev/null || echo 0)
  if [ "$_count" -lt "${PP_ROUTER_MIN:-1}" ]; then
    local _pad
    for _pad in $_enabled; do
      case " $_seen " in
        *" $_pad "*) ;;
        *)
          _validated="${_validated}${_pad}"$'\n'
          _seen="$_seen $_pad"
          _count=$((_count + 1))
        ;;
      esac
      [ "$_count" -ge "${PP_ROUTER_MIN:-1}" ] && break
    done
  fi

  # If still empty (no enabled set), fail-open to all enabled.
  if [ -z "$_validated" ]; then
    printf '%s\n' $_enabled
    return 0
  fi

  # Cap at MAX.
  printf '%s' "$_validated" | head -n "${PP_ROUTER_MAX:-3}"
}

# Internal: actually invoke the LLM. Returns raw model output (newline
# delimited). Caller is responsible for validation. Times out after
# PP_ROUTER_TIMEOUT_S; on timeout returns empty (caller falls open).
_pp_router_llm_call() {
  local _signals="$1" _tx="$2" _enabled="$3"
  command -v llm >/dev/null 2>&1 || { printf ''; return 0; }
  local _timeout_cmd=""
  if command -v timeout >/dev/null 2>&1; then
    _timeout_cmd="timeout ${PP_ROUTER_TIMEOUT_S:-8}"
  elif command -v gtimeout >/dev/null 2>&1; then
    _timeout_cmd="gtimeout ${PP_ROUTER_TIMEOUT_S:-8}"
  fi
  # Build the lens registry block.
  local _registry=""
  local _l
  for _l in $_enabled; do
    _registry="${_registry}- ${_l}"$'\n'
  done
  # Render prompt.
  local _prompt
  _prompt=$(. "$PP_ROOT/lib/prompt-loader.sh" >/dev/null 2>&1
            signals_json="$_signals" lens_registry="$_registry" \
              PP_ROUTER_MIN="${PP_ROUTER_MIN:-1}" PP_ROUTER_MAX="${PP_ROUTER_MAX:-3}" \
              pp_render_prompt router)
  $_timeout_cmd llm -m "${PP_ROUTER_MODEL:-gpt-5-mini}" -s "$_prompt" "Pick lenses." 2>/dev/null
}

# pp_router_surprise_inject <picked_space_separated> <not_picked_space_separated>
# Stdout: picked set, possibly appended with one random off-discipline lens.
# Deterministic when PP_RANDOM_SEED is set.
pp_router_surprise_inject() {
  local _picked="$1"
  local _candidates="$2"
  local _prob="${PP_ROUTER_SURPRISE_PROB:-0.2}"

  # Compare prob to a uniform random [0,1].
  local _roll
  if [ -n "${PP_RANDOM_SEED:-}" ]; then
    _roll=$(LC_ALL=C awk -v seed="$PP_RANDOM_SEED" 'BEGIN{srand(seed); printf "%.4f", rand()}')
  else
    _roll=$(LC_ALL=C awk 'BEGIN{srand(); printf "%.4f", rand()}')
  fi
  local _picked_lines
  _picked_lines=$(printf '%s\n' $_picked)
  if LC_ALL=C awk -v r="$_roll" -v p="$_prob" 'BEGIN{exit !(r<p)}' && [ -n "$_candidates" ]; then
    # Cap total.
    local _total
    _total=$(printf '%s\n' $_picked | grep -c '^.' 2>/dev/null || echo 0)
    if [ "$_total" -lt "${PP_ROUTER_SURPRISE_MAX_TOTAL:-4}" ]; then
      # Pick one candidate deterministically by index.
      local _cand_count _idx _surprise
      _cand_count=$(printf '%s\n' $_candidates | grep -c '^.' 2>/dev/null || echo 1)
      _idx=$(LC_ALL=C awk -v seed="${PP_RANDOM_SEED:-$$}" -v n="$_cand_count" 'BEGIN{srand(seed); printf "%d", int(rand()*n)+1}')
      _surprise=$(printf '%s\n' $_candidates | sed -n "${_idx}p")
      _picked_lines="${_picked_lines}"$'\n'"$_surprise"
    fi
  fi
  printf '%s\n' "$_picked_lines"
}
```
