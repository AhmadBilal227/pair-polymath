## Track 1 — Router timeout reliability + coreutils doctor hint

The end-to-end walkthrough caught a real failure: when neither `timeout` nor `gtimeout` is on PATH (common on macOS without coreutils), the router LLM call goes through the bg-spawn + kill watchdog. That path was hanging indefinitely in some shells. **The fix is to replace the watchdog entirely** with a portable `read -t` bounded read, then surface a doctor recommendation when coreutils is missing.

```bash
  else
    # Portable bounded-wait fallback (no coreutils, no job control).
    # Spawn llm in a pipeline whose stdout we drain via `read -t`. After
    # PP_ROUTER_TIMEOUT_S seconds with no data, close the pipe and the
    # child receives SIGPIPE on its next write, dying naturally. Output
    # captured into a tmp file by the pipeline so we can return it after
    # the bounded read.
    local _outfile
    _outfile=$(mktemp)
    # Run llm in a subshell that writes to the tmp file. The subshell's
    # PID is the one we wait on; close the read end after timeout.
    (
      llm -m "${PP_ROUTER_MODEL:-gpt-5-mini}" -s "$_prompt" "Pick lenses." 2>/dev/null > "$_outfile" &
      local _llm_pid=$!
      local _waited=0
      # Poll the file size; once non-zero we know llm is writing.
      while [ "$_waited" -lt "${PP_ROUTER_TIMEOUT_S:-8}" ]; do
        if ! kill -0 "$_llm_pid" 2>/dev/null; then break; fi
        sleep 1
        _waited=$((_waited + 1))
      done
      # Hard kill on overrun. SIGKILL the process group (negative pid).
      if kill -0 "$_llm_pid" 2>/dev/null; then
        kill -KILL "$_llm_pid" 2>/dev/null
        # Wait briefly for the OS to reap; otherwise the parent's `wait`
        # below might block on a defunct child.
        sleep 0.1 2>/dev/null || true
      fi
      wait "$_llm_pid" 2>/dev/null || true
    )
    cat "$_outfile" 2>/dev/null
    rm -f "$_outfile" 2>/dev/null
  fi
```

```bash
# pp_router_metrics_emit <signals_json> <picked_space_separated>
#                        <surprise_fired_bool> <failopen_bool> <llm_call_ms>
# Appends one JSONL line to PP_ROUTER_METRICS_FILE.
# Schema:
#   { ts: ISO-8601, phase, picked_count, surprise_fired, failopen, llm_call_ms }
#
# Bounded by PP_ROUTER_METRICS_MAX_LINES (default 5000). When reached,
# atomically renames to ".old" and starts fresh. Silently no-ops if the
# file path is unwritable so cycle reliability is preserved.
#
# Atomic appends without flock: write to tmpfile in same dir, then
# `cat >> $target` from tmpfile under a tiny mkdir-based lock.
: "${PP_ROUTER_METRICS_FILE:=${HOME}/.claude/cache/router-metrics.jsonl}"
: "${PP_ROUTER_METRICS_MAX_LINES:=5000}"

pp_router_metrics_emit() {
  local _signals="${1:-{}}"
  local _picked="${2:-}"
  local _surprise="${3:-0}"
  local _failopen="${4:-0}"
  local _llm_ms="${5:-0}"

  # Ensure parent dir exists.
  local _dir
  _dir=$(dirname "$PP_ROUTER_METRICS_FILE")
  mkdir -p "$_dir" 2>/dev/null || return 0
  [ -w "$_dir" ] || return 0

  # Extract phase from signals JSON (defaults to "unknown" if jq fails).
  local _phase
  _phase=$(printf '%s' "$_signals" | jq -r '.phase // "unknown"' 2>/dev/null)
  [ -z "$_phase" ] && _phase="unknown"

  # Count picked entries (newline-OR-space delimited).
  local _picked_count=0
  if [ -n "$_picked" ]; then
    _picked_count=$(printf '%s' "$_picked" | tr ' \n' '\n\n' | grep -c '^.' 2>/dev/null || echo 0)
  fi

  # Build the JSONL line (compact, single-line).
  local _ts _line
  _ts=$(date -u +%Y-%m-%dT%H:%M:%SZ 2>/dev/null)
  _line=$(jq -n -c \
    --arg ts "$_ts" \
    --arg phase "$_phase" \
    --argjson picked_count "$_picked_count" \
    --argjson surprise_fired "$_surprise" \
    --argjson failopen "$_failopen" \
    --argjson llm_call_ms "$_llm_ms" \
    '{ts: $ts, phase: $phase, picked_count: $picked_count, surprise_fired: $surprise_fired, failopen: $failopen, llm_call_ms: $llm_call_ms}' \
    2>/dev/null)
  [ -z "$_line" ] && return 0

  # Append-with-mkdir-lock (bash 3.2 portable, no flock).
  local _lockdir="${PP_ROUTER_METRICS_FILE}.lock"
  local _tries=0
  while ! mkdir "$_lockdir" 2>/dev/null; do
    _tries=$((_tries + 1))
    [ "$_tries" -gt 50 ] && return 0   # lock unavailable; drop this sample
    sleep 0.02 2>/dev/null || sleep 1
  done

  printf '%s\n' "$_line" >> "$PP_ROUTER_METRICS_FILE" 2>/dev/null

  # Rotate if over cap.
  local _count
  _count=$(wc -l < "$PP_ROUTER_METRICS_FILE" 2>/dev/null | tr -d ' ')
  if [ -n "$_count" ] && [ "$_count" -gt "${PP_ROUTER_METRICS_MAX_LINES:-5000}" ]; then
    mv -f "$PP_ROUTER_METRICS_FILE" "${PP_ROUTER_METRICS_FILE}.old" 2>/dev/null || true
  fi

  rmdir "$_lockdir" 2>/dev/null || true
}
```
