```bash
# pp_budget_remaining_pct
# Stdout: integer 0-100 representing remaining daily budget headroom.
# Reads today's pp-budget-YYYYMMDD.txt counter and divides into the
# PP_MAX_DAILY_CALLS cap. Clamps to [0, 100] so >cap stays at 0 (not
# negative). Single source of truth for "how much budget is left" —
# consumed by statusline line-1 pip, idle fallback messaging, and
# doctor budget-pressure check.
#
# No new lock; reads are inherently racy with concurrent budget_inc
# but we only need approximate state for UI/diagnostic purposes.
pp_budget_remaining_pct() {
  local _max="${PP_MAX_DAILY_CALLS:-10000}"
  [ "$_max" -le 0 ] 2>/dev/null && _max=10000
  local _file="${PP_CACHE_DIR:-${HOME}/.claude/cache}/pp-budget-$(date +%Y%m%d).txt"
  local _used=0
  [ -f "$_file" ] && _used=$(cat "$_file" 2>/dev/null) && [ -n "$_used" ] || _used=0
  [ "$_used" -eq "$_used" ] 2>/dev/null || _used=0
  local _pct=$(( 100 * (_max - _used) / _max ))
  [ "$_pct" -lt 0 ] && _pct=0
  [ "$_pct" -gt 100 ] && _pct=100
  printf '%d' "$_pct"
}
```

## Task 2: Tune `PP_DISPLAY_STALE_S` default

The current 600s (10 min) freshness gate is too tight given `PP_PARALLEL_INTERVAL_S=300s` (5 min) cycle interval. A cycle delayed by budget contention or LLM latency by one extra interval makes ALL caches stale at the same moment → all rotation slots skip → fall through to tip → user sees no lens observations.

New default: `max(900, 3 * PP_PARALLEL_INTERVAL_S)`. Default config gives 900s (15 min). Users who bump `PP_PARALLEL_INTERVAL_S` get a proportionally larger gate.

```bash
  local _pct

  [ -z "$_pct" ] && _pct=100
  local _used=$(( 100 - _pct ))
  if [ "$_used" -ge 95 ]; then

    return 1
  fi
  _pp_doctor_green "budget pressure" "${_pct}% remaining"
  return 0
}
```
