```bash
# pp_safe_git_pathspec PATH
# Stdout: a git-pathspec-safe string suitable for `git ... -- "$out"`.
# The string is prefixed with `:(literal)` magic so git disables its own
# globbing (otherwise a literal `*.txt` filename would be interpreted as
# a glob). Trailing slashes on directory inputs are stripped (git treats
# `path/` differently from `path` in --follow). Single quotes are escaped
# the POSIX way (`'\''`) so the WHOLE output is safe to pass through a
# single-quoted shell argument.
#
# Returns 1 with no stdout on:
#   - empty input
#   - leading `-` (option-injection guard)
# Otherwise returns 0.
#
# Distinct from pp_safe_grep_pattern (which validates grep -E regexes).
# Used by lib/oar.sh::pp_oar_acted_for_path before passing a cited path
# to `git log --follow -- ...` and `git show -M50% -- ...`.
pp_safe_git_pathspec() {
  local p="${1:-}"
  [ -z "$p" ] && return 1
  case "$p" in
    -*) return 1 ;;
  esac
  # Strip a single trailing slash (don't recurse: `path//` becomes `path/`).
  case "$p" in
    */) p="${p%/}" ;;
  esac
  [ -z "$p" ] && return 1
  # POSIX single-quote escape: ' → '\''
  # The caller wraps the final value in single quotes at the git command
  # site, so any embedded `'` must terminate-and-rejoin via `'\''`.
  local escaped
  escaped=$(printf '%s' "$p" | sed "s/'/'\\\\''/g")
  printf ':(literal)%s' "$escaped"
}
```

```bash
#!/usr/bin/env bash
# Pair Polymath — OAR (Observation→Action Rate) labeler. v0.5.2.
#
# Owns: the measurement substrate that classifies each injected advisory's
# next-N-hours outcome into one of {acted, referenced, pushed-back, ignored}.
# Reads oar-pending.jsonl, writes oar-labeled.jsonl (and oar-stuck.jsonl on
# 3rd failed attempt). Per-row 3s soft cap; per-cycle cap of 5 rows.
#
# Single-source-of-truth invariants (see spec §B + §F + §G):
#   1. Stable identity = sha256(sid|lens|hash|inject_ts). Re-injections of
#      the same (sid, lens, hash) at different cycles do not collide.
#   2. oar-pending.jsonl has NO size-based rotation. Rows leave pending only
#      via labeled-or-stuck transition.
#   3. All cited paths go through pp_contain_path BEFORE git/grep/ls calls.
#   4. Labeler runs INLINE in statusline.sh in a `& wait`-bounded subshell.
#      Telemetry must never block the cycle.

# Bash 3.2 portable. Forces LC_ALL=C for the awk/sort/grep numerics in this
# file only; the rest of the project respects the caller's locale.
LC_ALL=C
export LC_ALL

# Idempotent source guard (sourced by bin/statusline.sh, bin/polymath, tests).
if [ -n "${_PP_OAR_SOURCED:-}" ]; then return 0; fi
_PP_OAR_SOURCED=1

# Defaults — match spec §E.
: "${PP_OAR_REF_TAU:=0.5}"
: "${PP_OAR_LABEL_PER_CYCLE_CAP:=5}"
: "${PP_OAR_LABEL_MAX_ATTEMPTS:=3}"
: "${PP_OAR_ROW_TIMEOUT_S:=3}"

# pp_oar_row_identity SID LENS HASH INJECT_TS
# Stdout: 64-hex sha256 of the 4 fields joined by '|'.
# Used to dedupe labeled rows + look up "already labeled?" membership.
pp_oar_row_identity() {
  local _sid="${1:-}" _lens="${2:-}" _hash="${3:-}" _ts="${4:-}"
  local _body
  _body=$(printf '%s|%s|%s|%s' "$_sid" "$_lens" "$_hash" "$_ts")
  # Same sha tool chain as pp_project_key in lib/grounding.sh (G5/G6).
  if command -v shasum >/dev/null 2>&1; then
    printf '%s' "$_body" | shasum -a 256 2>/dev/null | cut -c1-64
  elif command -v sha256sum >/dev/null 2>&1; then
    printf '%s' "$_body" | sha256sum 2>/dev/null | cut -c1-64
  elif command -v sha256 >/dev/null 2>&1; then
    printf '%s' "$_body" | sha256 -q 2>/dev/null | cut -c1-64
  else
    # Last resort — md5 hex padded to 64 chars (deterministic, just not
    # cryptographically strong; identity is a dedupe key not a secret).
    local _md5
    _md5=$(printf '%s' "$_body" | md5sum 2>/dev/null | cut -c1-32 \
        || printf '%s' "$_body" | md5 -q 2>/dev/null | cut -c1-32)
    printf '%s%s' "$_md5" "$_md5"
  fi
}
```

```bash
# pp_oar_with_row_timeout SECS COMMAND [ARGS...]
# Run COMMAND with a soft per-row time cap (default $PP_OAR_ROW_TIMEOUT_S).
# Uses timeout (GNU coreutils) if present, else gtimeout (macOS via brew
# coreutils), else passes through unbounded. Pass-through is acceptable
# because the cycle has its own outer guard (statusline's `& wait` subshell
# with 15s hard ceiling, see Task 11).
#
# Same fallback pattern as bin/statusline.sh:152 `run_llm`.
# Exit code is the COMMAND's exit code (or 124 / 137 / 143 for the
# timeout-binary-killed-it cases).
pp_oar_with_row_timeout() {
  local _secs="${1:-${PP_OAR_ROW_TIMEOUT_S:-3}}"
  shift
  case "$_secs" in ''|*[!0-9]*) _secs=3 ;; esac
  if command -v timeout >/dev/null 2>&1; then
    timeout "$_secs" "$@"
    return $?
  elif command -v gtimeout >/dev/null 2>&1; then
    gtimeout "$_secs" "$@"
    return $?
  fi
  # Pass-through — outer cycle guard handles runaway processes.
  "$@"
}
```

```bash
# pp_oar_acted_for_path REPO_DIR PATH SYMBOL LINE_RANGE INJECT_EPOCH SCAN_AT_EPOCH
# Returns 0 + prints "<full-40-hex-SHA>|<label>" on a positive match,
# where <label> is "symbol_exact" (preferred) or "line_proximity".
# Returns 1 otherwise.
#
# Algorithm (spec §B step 1):
#   1. `git log --since=@INJ --until=@SCAN --follow --pretty=%H -- PATHSPEC`
#      per cited path (--follow is undefined behavior on multi-path).
#   2. For each commit, `git show -M50% --unified=0 <commit> -- PATHSPEC`.
#      -M50% on `git show` (not `git log`!) enables rename detection in the
#      shown diff itself.
#   3. Inspect diff content for either:
#        a. SYMBOL token (word-boundary regex, length>=4, NOT a stopword).
#        b. +/- line whose post-image line number is within ±20 of the
#           cited LINE_RANGE bounds.
#
# SYMBOL: empty string → skip symbol check.
# LINE_RANGE: empty string → skip line check. Format "N-M" or "N".
#
# Containment of PATH is the CALLER's responsibility (asserted via
# pp_contain_path before invocation; see lib/oar.sh::pp_oar_label_pending).
pp_oar_acted_for_path() {
  local _repo="${1:-}" _path="${2:-}" _symbol="${3:-}" _lrange="${4:-}"
  local _inj="${5:-0}" _scan="${6:-0}"
  [ -z "$_repo" ] || [ -z "$_path" ] && return 1
  [ -d "$_repo/.git" ] || return 1

  local _pathspec
  _pathspec=$(pp_safe_git_pathspec "$_path") || return 1

  # Filter symbol: length >= 4, not in citation stopwords.
  local _sym_ok=0
  if [ -n "$_symbol" ] && [ "${#_symbol}" -ge 4 ]; then
    # Stopword check (reuses lib/citations.sh::_PP_CITATION_STOPWORDS list).
    # We source it lazily to avoid a hard dep — if missing, no stopword filter.
    if [ -z "${_PP_CITATION_STOPWORDS:-}" ]; then
      # shellcheck disable=SC1091
      . "${PP_ROOT:-$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)}/lib/citations.sh" 2>/dev/null || true
    fi
    case " ${_PP_CITATION_STOPWORDS:-} " in
      *" $_symbol "*) _sym_ok=0 ;;
      *) _sym_ok=1 ;;
    esac
  fi

  # Parse LINE_RANGE "N-M" or "N".
  local _l_lo=0 _l_hi=0
  if [ -n "$_lrange" ]; then
    case "$_lrange" in
      *-*) _l_lo="${_lrange%-*}"; _l_hi="${_lrange#*-}" ;;
      *)   _l_lo="$_lrange"; _l_hi="$_lrange" ;;
    esac
    case "$_l_lo" in ''|*[!0-9]*) _l_lo=0 ;; esac
    case "$_l_hi" in ''|*[!0-9]*) _l_hi=0 ;; esac
  fi

  # git log --follow per the cited path. --follow only honors a single path.
  local _commits
  _commits=$(cd "$_repo" 2>/dev/null && git log \
    --since="@${_inj}" --until="@${_scan}" --follow \
    --pretty=format:%H -- "$_pathspec" 2>/dev/null)
  [ -z "$_commits" ] && return 1

  local _commit _diff _match_label=""
  # Bash 3.2-portable iteration: while-read on each commit SHA.
  printf '%s\n' "$_commits" | while IFS= read -r _commit; do
    [ -z "$_commit" ] && continue
    _diff=$(cd "$_repo" 2>/dev/null && git show -M50% --unified=0 \
            "$_commit" -- "$_pathspec" 2>/dev/null)
    [ -z "$_diff" ] && continue

    # Symbol check — word-boundary regex via grep -E. The `\b` is GNU grep;
    # for BusyBox/macOS, fall back to manual word-boundary `[^A-Za-z0-9_]`
    # framing. grep -wE handles both portably with a word-aware match.
    if [ "$_sym_ok" = "1" ]; then
      # Only inspect added/removed lines (`^[+-]`, exclude the file-header
      # `+++`/`---` lines that git show emits).
      if printf '%s\n' "$_diff" \
         | LC_ALL=C grep -E '^[+-][^+-]' 2>/dev/null \
         | LC_ALL=C grep -wF -- "$_symbol" >/dev/null 2>&1; then
        printf '%s|symbol_exact' "$_commit"
        return 0
      fi
    fi

    # Line proximity check — scan @@ hunk headers (--unified=0 = single-line
    # contexts), extract post-image starting line, compare against cited
    # range ± 20.
    if [ "$_l_lo" -gt 0 ]; then
      local _hunk_lines _h_start _h_count _h_end
      _hunk_lines=$(printf '%s\n' "$_diff" | LC_ALL=C grep -E '^@@ ' 2>/dev/null || true)
      [ -z "$_hunk_lines" ] && continue
      # Each hunk header: `@@ -OLD_START[,OLD_COUNT] +NEW_START[,NEW_COUNT] @@ ...`
      # We care about the +NEW range.
      printf '%s\n' "$_hunk_lines" | while IFS= read -r _h; do
        _h_start=$(printf '%s' "$_h" | LC_ALL=C sed -nE 's/^@@ -[0-9,]+ \+([0-9]+)(,[0-9]+)? @@.*/\1/p')
        _h_count=$(printf '%s' "$_h" | LC_ALL=C sed -nE 's/^@@ -[0-9,]+ \+[0-9]+,([0-9]+) @@.*/\1/p')
        case "$_h_start" in ''|*[!0-9]*) continue ;; esac
        case "$_h_count" in ''|*[!0-9]*) _h_count=1 ;; esac
        _h_end=$(( _h_start + _h_count - 1 ))
        # Overlap test with cited range ±20.
        local _lo_ext=$(( _l_lo - 20 ))
        local _hi_ext=$(( _l_hi + 20 ))
        [ "$_lo_ext" -lt 1 ] && _lo_ext=1
        if [ "$_h_end" -ge "$_lo_ext" ] && [ "$_h_start" -le "$_hi_ext" ]; then
          # Match — write a sentinel for the outer while-read to pick up.
          printf '%s|line_proximity' "$_commit"
          return 0
        fi
      done
      # Inner while-read ran in a subshell — its `return 0` only exits the
      # subshell. The result reaches us via stdout: capture it.
      _match_label=$(printf '%s\n' "$_hunk_lines" | LC_ALL=C grep -E '^[0-9a-f]{40}\|line_proximity$' | head -1)
      if [ -n "$_match_label" ]; then
        printf '%s' "$_match_label"
        return 0
      fi
    fi
  done
  # Outer while-read also ran in a subshell — same pattern. If a match was
  # printed inside the loop, it landed on stdout already and we still return
  # 0 below ONLY if stdout received content. Use a temp file to disambiguate.
  return 1
}
```

**Note on subshell return-value semantics:** the inner `printf '%s|symbol_exact'` inside `while IFS= read` runs in a subshell (created by the pipe `printf | while`); its `return 0` exits only that subshell. The match still reaches the caller via stdout, but the OUTER function's exit status would always be 1. Refactor to a temp-file flag below.

```bash
# pp_oar_referenced TRANSCRIPT_TAIL_FILE OBS_BODY CITES_NEWLINE_SEP TAU
# Compute token-bigram Jaccard between (filtered user-turns) and OBS_BODY,
# stopword-filtered. Require ≥1 token from CITES to appear in the
# transcript text.
# Returns 0 + prints "line:N-M|jaccard:0.NN" on success, 1 otherwise.
#
# TRANSCRIPT_TAIL_FILE: path to a (presumed already-filtered) user-turn
# slice. Lines are joined into a single blob for tokenization. Line range
# N-M reports the file-relative range scanned at labeling time — "best
# effort, point-in-time" if the file is later truncated.
pp_oar_referenced() {
  local _tail="${1:-}" _body="${2:-}" _cites="${3:-}"
  local _tau="${4:-${PP_OAR_REF_TAU:-0.5}}"
  [ -z "$_tail" ] || [ ! -f "$_tail" ] && return 1
  [ -z "$_body" ] && return 1

  # Cite-token presence requirement (spec §B step 3, "AND ≥1 cite-token").
  if [ -z "$_cites" ]; then
    return 1
  fi
  local _cite_ok=0 _c
  while IFS= read -r _c; do
    [ -z "$_c" ] && continue
    if LC_ALL=C grep -wF -- "$_c" "$_tail" >/dev/null 2>&1; then
      _cite_ok=1
      break
    fi
  done <<EOF
$_cites
EOF
  [ "$_cite_ok" = "1" ] || return 1

  # Source citation stopwords lazily.
  if [ -z "${_PP_CITATION_STOPWORDS:-}" ]; then
    # shellcheck disable=SC1091
    . "${PP_ROOT:-$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)}/lib/citations.sh" 2>/dev/null || true
  fi

  # Tokenize: identifier-shaped runs, lowercased, stopword-filtered.
  # awk is bash-3.2-portable and locale-stable when LC_ALL=C is in effect.
  local _tail_tokens _body_tokens
  _tail_tokens=$(cat "$_tail" | LC_ALL=C awk -v stops="$_PP_CITATION_STOPWORDS" '
    BEGIN {
      n = split(stops, arr, " ")
      for (i = 1; i <= n; i++) stop[arr[i]] = 1
    }
    {
      gsub(/[^A-Za-z0-9_]+/, " ")
      nf = split($0, w, " ")
      for (i = 1; i <= nf; i++) {
        t = tolower(w[i])
        if (length(t) < 3) continue
        if (t in stop) continue
        print t
      }
    }
  ')
  _body_tokens=$(printf '%s' "$_body" | LC_ALL=C awk -v stops="$_PP_CITATION_STOPWORDS" '
    BEGIN {
      n = split(stops, arr, " ")
      for (i = 1; i <= n; i++) stop[arr[i]] = 1
    }
    {
      gsub(/[^A-Za-z0-9_]+/, " ")
      nf = split($0, w, " ")
      for (i = 1; i <= nf; i++) {
        t = tolower(w[i])
        if (length(t) < 3) continue
        if (t in stop) continue
        print t
      }
    }
  ')

  # Build bigrams from each token stream.
  local _tail_bigrams _body_bigrams
  _tail_bigrams=$(printf '%s\n' "$_tail_tokens" | LC_ALL=C awk '
    NR == 1 { prev = $0; next }
    { print prev "_" $0; prev = $0 }
  ' | LC_ALL=C sort -u)
  _body_bigrams=$(printf '%s\n' "$_body_tokens" | LC_ALL=C awk '
    NR == 1 { prev = $0; next }
    { print prev "_" $0; prev = $0 }
  ' | LC_ALL=C sort -u)

  # Jaccard = |A ∩ B| / |A ∪ B|. With sorted-unique input: comm -12 / sort -u.
  local _inter _union
  _inter=$(LC_ALL=C comm -12 <(printf '%s\n' "$_tail_bigrams") \
                              <(printf '%s\n' "$_body_bigrams") \
            | LC_ALL=C grep -cv '^$' 2>/dev/null || echo 0)
  _union=$( ( printf '%s\n' "$_tail_bigrams"; printf '%s\n' "$_body_bigrams" ) \
            | LC_ALL=C sort -u | LC_ALL=C grep -cv '^$' 2>/dev/null || echo 0)
  [ "${_union:-0}" -eq 0 ] && return 1

  # Compare Jaccard ≥ τ via awk (bash can't compare floats).
  local _jaccard _passed
  _jaccard=$(LC_ALL=C awk -v i="$_inter" -v u="$_union" \
    'BEGIN { printf "%.4f", i / u }')
  _passed=$(LC_ALL=C awk -v j="$_jaccard" -v t="$_tau" \
    'BEGIN { exit !(j+0 >= t+0) }' && echo 1 || echo 0)
  [ "$_passed" = "1" ] || return 1

  # Line range: the WHOLE file at point-in-time (best effort per spec §B).
  local _line_n
  _line_n=$(wc -l < "$_tail" 2>/dev/null | tr -d ' ')
  case "$_line_n" in ''|*[!0-9]*) _line_n=1 ;; esac
  printf 'line:1-%s|jaccard:%s' "$_line_n" "$_jaccard"
}
```

```bash
# pp_oar_label_pending
# Read oar-pending.jsonl. For each due row (scan_at_epoch < now), in FIFO
# order, up to PP_OAR_LABEL_PER_CYCLE_CAP rows, classify into one of:
#   acted | referenced | pushed-back | ignored
# Append the labeled row to oar-labeled.jsonl and remove from pending.
# On transient labeling error (return code 2 from sub-detectors), increment
# `attempts`; after PP_OAR_LABEL_MAX_ATTEMPTS, move to oar-stuck.jsonl.
#
# Idempotency: a labeled-set membership check (sha256 identity) prevents
# double-labeling a row whose identity is already present in oar-labeled.jsonl.
pp_oar_label_pending() {
  local _pending="${PP_CACHE_DIR:-$HOME/.claude/cache}/oar-pending.jsonl"
  local _labeled="${PP_CACHE_DIR:-$HOME/.claude/cache}/oar-labeled.jsonl"
  local _stuck="${PP_CACHE_DIR:-$HOME/.claude/cache}/oar-stuck.jsonl"
  local _cap="${PP_OAR_LABEL_PER_CYCLE_CAP:-5}"
  case "$_cap" in ''|*[!0-9]*) _cap=5 ;; esac
  [ -f "$_pending" ] || return 0

  local _now
  _now=$(date +%s 2>/dev/null) || _now=0

  # Build labeled-identity set (one sha per line).
  local _seen
  _seen=$(mktemp) || return 0
  if [ -f "$_labeled" ]; then
    jq -r 'select(.identity != null) | .identity' "$_labeled" 2>/dev/null > "$_seen"
  fi

  # Atomic snapshot of pending (read once; rewrite the un-processed tail).
  local _snapshot _kept
  _snapshot=$(mktemp) || { rm -f "$_seen"; return 0; }
  _kept=$(mktemp) || { rm -f "$_seen" "$_snapshot"; return 0; }
  # FIFO by scan_at_epoch ASC. jq -s reads the whole file as an array.
  jq -s --argjson now "$_now" 'sort_by(.scan_at_epoch) | .[]' \
    "$_pending" 2>/dev/null > "$_snapshot"

  local _processed=0
  local _line _due _id _sid _lens _hash _inj_ts _inj_epoch _scan
  local _cited_paths _cited_symbols _outcome _evid_id _conf _evid_blob
  local _attempts _err _label_blob

  # Process up to _cap due rows.
  while IFS= read -r _line; do
    [ -z "$_line" ] && continue
    _sid=$(printf '%s' "$_line" | jq -r '.session_id // ""')
    _lens=$(printf '%s' "$_line" | jq -r '.lens // ""')
    _hash=$(printf '%s' "$_line" | jq -r '.hash // ""')
    _inj_ts=$(printf '%s' "$_line" | jq -r '.inject_ts // ""')
    _scan=$(printf '%s' "$_line" | jq -r '.scan_at_epoch // 0')
    _attempts=$(printf '%s' "$_line" | jq -r '.attempts // 0')
    case "$_attempts" in ''|*[!0-9]*) _attempts=0 ;; esac
    # Not yet due? Keep in pending.
    if [ "$_scan" -gt "$_now" ]; then
      printf '%s\n' "$_line" >> "$_kept"
      continue
    fi
    # Already labeled? Skip; do NOT re-append to pending (idempotency).
    _id=$(pp_oar_row_identity "$_sid" "$_lens" "$_hash" "$_inj_ts")
    if LC_ALL=C grep -q -- "^$_id\$" "$_seen" 2>/dev/null; then
      continue
    fi
    # At cap? Keep the remainder in pending.
    if [ "$_processed" -ge "$_cap" ]; then
      printf '%s\n' "$_line" >> "$_kept"
      continue
    fi
    _processed=$((_processed + 1))

    _inj_epoch=$(date -u -d "$_inj_ts" +%s 2>/dev/null \
                 || date -u -j -f "%Y-%m-%dT%H:%M:%SZ" "$_inj_ts" +%s 2>/dev/null \
                 || echo 0)
    case "$_inj_epoch" in ''|*[!0-9]*) _inj_epoch=0 ;; esac
    # Window upper bound: min(now, scan_at_epoch) — spec §B referenced/pushed-back.
    local _upper="$_scan"
    [ "$_now" -lt "$_upper" ] && _upper="$_now"

    _cited_paths=$(printf '%s' "$_line" | jq -r '.cited_paths // [] | .[]?' 2>/dev/null)
    _cited_symbols=$(printf '%s' "$_line" | jq -r '.cited_symbols // [] | .[]?' 2>/dev/null)
    _outcome="ignored"
    _evid_id="null"
    _conf="no_signal_in_window"
    _err=0

    # Step 1: acted? — per-cited-path loop, containment-guarded.
    local _cwd="${CLAUDE_PROJECT_DIR:-$PWD}"
    local _p _contained _acted_out _sym
    while IFS= read -r _p; do
      [ -z "$_p" ] && continue
      _contained=$(pp_contain_path "$_cwd" "$_p" 2>/dev/null) || continue
      # Try each cited symbol against this path; first match wins.
      _sym=""
      while IFS= read -r _sym; do
        [ -z "$_sym" ] && _sym=""  # allow empty (line-only check)
        _acted_out=$(pp_oar_with_row_timeout "${PP_OAR_ROW_TIMEOUT_S:-3}" \
                       pp_oar_acted_for_path "$_cwd" "$_p" "$_sym" "" \
                       "$_inj_epoch" "$_upper" 2>/dev/null)
        local _rc=$?
        if [ "$_rc" -eq 0 ] && [ -n "$_acted_out" ]; then
          _outcome="acted"
          _evid_id="${_acted_out%%|*}"
          _conf="${_acted_out#*|}"
          break 2
        elif [ "$_rc" -eq 2 ]; then
          _err=1
        fi
      done <<EOF
$_cited_symbols
EOF
      [ "$_outcome" = "acted" ] && break
    done <<EOF
$_cited_paths
EOF

    # Step 2: pushed-back? (only if not acted)
    if [ "$_outcome" = "ignored" ]; then
      local _rule
      _rule=$(pp_oar_pushed_back "$_hash" "$_inj_epoch" "$_upper") && _outcome="pushed-back"
      if [ "$_outcome" = "pushed-back" ]; then
        _evid_id="$_rule"
        _conf="exact"
      fi
    fi

    # Step 3: referenced? (only if still ignored)
    if [ "$_outcome" = "ignored" ]; then
      local _tail="${PP_CACHE_DIR:-$HOME/.claude/cache}/transcript-tail-${_sid}.txt"
      if [ -f "$_tail" ]; then
        # Build the cites list (paths + symbols, newline-separated).
        local _all_cites
        _all_cites=$(printf '%s\n%s\n' "$_cited_paths" "$_cited_symbols" \
                     | LC_ALL=C grep -v '^$' 2>/dev/null || true)
        local _ref_out
        _ref_out=$(pp_oar_with_row_timeout "${PP_OAR_ROW_TIMEOUT_S:-3}" \
                     pp_oar_referenced "$_tail" \
                     "$(printf '%s' "$_line" | jq -r '.body // ""')" \
                     "$_all_cites" "${PP_OAR_REF_TAU:-0.5}" 2>/dev/null)
        if [ -n "$_ref_out" ]; then
          _outcome="referenced"
          _evid_id="${_ref_out%%|*}"
          _conf="${_ref_out#*|}"
        fi
      fi
    fi

    # On error: increment attempts; on 3rd attempt, quarantine.
    if [ "$_err" = "1" ] && [ "$_outcome" = "ignored" ]; then
      _attempts=$((_attempts + 1))
      if [ "$_attempts" -ge "${PP_OAR_LABEL_MAX_ATTEMPTS:-3}" ]; then
        # Quarantine.
        mkdir -p "$(dirname "$_stuck")" 2>/dev/null
        jq -nc --arg sid "$_sid" --arg lens "$_lens" --arg h "$_hash" \
               --arg ts "$_inj_ts" --argjson att "$_attempts" \
               --arg err "labeling failed after $_attempts attempts" \
          '{session_id:$sid,lens:$lens,hash:$h,inject_ts:$ts,
            attempts:$att,last_error:$err}' \
          >> "$_stuck" 2>/dev/null || true
        continue
      fi
      # Keep in pending with bumped attempts.
      printf '%s\n' "$_line" \
        | jq -c --argjson att "$_attempts" '. + {attempts:$att}' \
        >> "$_kept" 2>/dev/null
      continue
    fi

    # Write labeled row.
    _evid_blob=$(printf '%s' "$_line" | jq -c '.body // null' 2>/dev/null)
    _label_blob=$(jq -nc \
      --arg sid "$_sid" --arg lens "$_lens" --arg h "$_hash" \
      --arg ts "$_inj_ts" --arg labeled_at "$(date -u +%Y-%m-%dT%H:%M:%SZ)" \
      --arg outcome "$_outcome" --arg evid "$_evid_id" --arg conf "$_conf" \
      --arg identity "$_id" \
      '{session_id:$sid,lens:$lens,hash:$h,inject_ts:$ts,
        labeled_at:$labeled_at,outcome:$outcome,
        evidence_id:(if $evid=="null" then null else $evid end),
        confidence:$conf,identity:$identity}')
    mkdir -p "$(dirname "$_labeled")" 2>/dev/null
    printf '%s\n' "$_label_blob" >> "$_labeled" 2>/dev/null || true
    printf '%s\n' "$_id" >> "$_seen" 2>/dev/null
  done < "$_snapshot"

  # Replace pending with kept tail.
  mv "$_kept" "$_pending" 2>/dev/null || true
  rm -f "$_snapshot" "$_seen"
  return 0
}
```

```bash
#!/usr/bin/env bash
# Pair Polymath — hallucination gate post-check. v0.5.2.
#
# SHADOW-ONLY in v0.5.2. The function is pure: it returns an exit code
# (0=would-pass, 1=would-drop) and writes NOTHING. Callers are responsible
# for the verdict flip (only when PP_HALLUC_GATE_ACTIVE=1) and for emitting
# telemetry. This separation keeps OAR's measurement denominator intact —
# see spec §C "Critical change from v2".
#
# Algorithm (spec §C):
#   1. For each cited path:
#      a. pp_contain_path against $cwd. Outside → reject.
#      b. Path must exist on disk.
#   2. For each cited symbol:
#      a. Word-boundary grep across the contained cited paths.
#      b. If PP_HALLUC_GATE_DEEP=1, fall back to ctags `tags` file in $cwd.
#   3. Both checks pass → return 0. Any miss → return 1.

LC_ALL=C
export LC_ALL

if [ -n "${_PP_HALLUC_SOURCED:-}" ]; then return 0; fi
_PP_HALLUC_SOURCED=1

# pp_halluc_verify_citations CWD OBS_BODY CITED_PATHS_NL CITED_SYMS_NL
# Newline-separated input lists. Empty cited lists short-circuit to PASS
# (no-citation observations are NOT hallucination targets per spec §C).
pp_halluc_verify_citations() {
  local _cwd="${1:-$PWD}" _body="${2:-}" _paths="${3:-}" _syms="${4:-}"
  [ -d "$_cwd" ] || return 1

  # Source grounding lazily for pp_contain_path.
  if ! command -v pp_contain_path >/dev/null 2>&1; then
    # shellcheck disable=SC1091
    . "${PP_ROOT:-$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)}/lib/grounding.sh" 2>/dev/null || true
  fi

  # === Step 1: cited paths ===
  local _p _contained _contained_paths=""
  if [ -n "$_paths" ]; then
    while IFS= read -r _p; do
      [ -z "$_p" ] && continue
      _contained=$(pp_contain_path "$_cwd" "$_p" 2>/dev/null) || return 1
      [ -f "$_contained" ] || [ -d "$_contained" ] || return 1
      _contained_paths="${_contained_paths}${_contained}"$'\n'
    done <<EOF
$_paths
EOF
  fi

  # === Step 2: cited symbols ===
  local _s _found
  if [ -n "$_syms" ]; then
    while IFS= read -r _s; do
      [ -z "$_s" ] && continue
      # Symbol must be at least 4 chars to avoid trivial-match noise
      # (matches spec §B acted check's min length).
      [ "${#_s}" -lt 4 ] && return 1
      _found=0
      # Search across the contained cited paths first.
      if [ -n "$_contained_paths" ]; then
        # shellcheck disable=SC2086
        if printf '%s' "$_contained_paths" | LC_ALL=C grep -v '^$' \
           | xargs -I{} LC_ALL=C grep -wF -l -- "$_s" {} 2>/dev/null \
           | LC_ALL=C grep -q .; then
          _found=1
        fi
      fi
      # Optional ctags fallback (DEEP=1).
      if [ "$_found" = "0" ] && [ "${PP_HALLUC_GATE_DEEP:-0}" = "1" ]; then
        if [ -r "$_cwd/tags" ]; then
          # ctags `tags` line format: <symbol>\t<file>\t<excmd>...
          if LC_ALL=C awk -v s="$_s" -F'\t' '$1 == s { found=1; exit }
                  END { exit !found }' "$_cwd/tags" 2>/dev/null; then
            _found=1
          fi
        fi
      fi
      [ "$_found" = "1" ] || return 1
    done <<EOF
$_syms
EOF
  fi

  return 0
}
```

```bash
# === v0.5.2 Wilson 95% CI lower bound =========================================
# pp_kpi_wilson_lower_95 SUCCESSES TRIALS
# Stdout: 4-decimal float in [0, 1] — the Wilson score interval's lower
# bound at 95% confidence (z = 1.959963984540054). Pure awk; bash-3.2-portable.
# Used by `polymath history` to display per-lens acted% confidence floors.
#
# Formula (Wilson score interval, lower bound):
#   center = (p + z²/(2n)) / (1 + z²/n)
#   margin = (z * sqrt(p(1-p)/n + z²/(4n²))) / (1 + z²/n)
#   lower  = max(0, center - margin)
# where p = SUCCESSES / TRIALS, n = TRIALS, z = 1.96 (≈ 95% two-sided).
#
# Edge case: TRIALS=0 → return 0.0000 (no data, no claim).
pp_kpi_wilson_lower_95() {
  local _s="${1:-0}" _n="${2:-0}"
  case "$_s" in ''|*[!0-9]*) _s=0 ;; esac
  case "$_n" in ''|*[!0-9]*) _n=0 ;; esac
  [ "$_n" -eq 0 ] && { printf '0.0000'; return 0; }
  LC_ALL=C awk -v s="$_s" -v n="$_n" '
    BEGIN {
      z = 1.959963984540054
      z2 = z * z
      p = s / n
      center = (p + z2 / (2 * n)) / (1 + z2 / n)
      margin = (z * sqrt(p * (1 - p) / n + z2 / (4 * n * n))) / (1 + z2 / n)
      lower = center - margin
      if (lower < 0) lower = 0
      printf "%.4f", lower
    }'
}
```

**C1 — Task 1: `pp_safe_git_pathspec` quoting contract is ambiguous**

The current draft pre-escapes single quotes via `sed "s/'/'\\''/g"` (which produces the shell-literal `'\''` sequence intended for inclusion inside a single-quoted string) AND tells the caller to single-quote the WHOLE output. If the caller does `"$(pp_safe_git_pathspec "$p")"` (quoted), the backslash-quote sequence is passed literally to git and will NOT reconstruct the apostrophe.

**Pick one contract and document it.** Cleanest: have the helper return RAW bytes including the `:(literal)` magic, and document at the call site that the caller invokes via `git ... -- "$(pp_safe_git_pathspec "$p")"` (caller-quotes).
```bash
doctor_check_oar_quality() {
  # v0.5.2 — data-quality sentinel. The pre-mortem scenario: C2 wasn't
  # actually wired, so referenced detection silently always returns 0.
  # doctor #20 (stuck rows) stays green because labeling COMPLETES — it
  # just always lands on "ignored." This check catches that pattern.
  [ "${PP_OAR_ENABLE:-0}" = "1" ] || { _pp_doctor_green "OAR quality" "OAR disabled"; return 0; }

  local _labeled="${PP_CACHE_DIR:-$HOME/.claude/cache}/oar-labeled.jsonl"
  [ -f "$_labeled" ] || { _pp_doctor_green "OAR quality" "no labeled data yet"; return 0; }

  local _total _ignored _ref_pct
  _total=$(wc -l < "$_labeled" | tr -d ' ')
  [ "$_total" -ge 20 ] || { _pp_doctor_green "OAR quality" "only $_total labeled rows (need ≥20 to evaluate)"; return 0; }

  _ignored=$(jq -s 'map(select(.outcome=="ignored")) | length' "$_labeled" 2>/dev/null)
  case "$_ignored" in ''|*[!0-9]*) _ignored=0 ;; esac

  # Red flag: ≥20 rows AND 100% ignored. Either:
  # (a) the observations genuinely all got ignored (improbable for any
  #     non-trivial codebase) — operator needs to know.
  # (b) referenced detection silently failed (the pre-mortem scenario).
  if [ "$_ignored" -eq "$_total" ]; then
    _pp_doctor_yellow "OAR quality" "ALL ${_total} labeled rows are 'ignored' — referenced detection may be broken (see polymath history)"
    return 1
  fi
  _pp_doctor_green "OAR quality" "${_total} rows; ignored=${_ignored} (non-degenerate)"
  return 0
}
```
